// =============================================================================
// RANK CHALLENGE — F12 CONSOLE TEST (v2)
// Ctrl+A copy TOAN FILE → dan Console game
//
// Loi ban gap: invalid_slot
//   → leaderboard KHONG co field "slot", tool cu bia slot 11/14/15 = SAI
//   → Win that cua ban: board=na, target_slot=2 hoac 3
//   → Script nay: chi dung slot that; neu thieu → probe slot 2..10
// =============================================================================

(async () => {
  // ---------- CONFIG ----------
  const ACCESS_TOKEN = "PASTE_BEARER_TOKEN_HERE";
  const CHARACTER_ID = "PASTE_CHARACTER_UUID_HERE";
  // Preview win that: board_code "na"
  const BOARD_CODE = "na"; // lk | tc | kd | na | ht | lh
  // ----------------------------

  const MAX_FIGHTS = 5;
  const DELAY_MS = 1500;
  const PROBE_SLOTS = [2, 3, 4, 5, 6, 7, 8, 9, 10]; // gan rank 1
  const API_KEY = "sb_publishable_vNnNBJooTMczVrWP7qCnhA_479q9nKB";
  const BASE = "https://jeassefmlprfnlszgvbs.supabase.co";

  if (!ACCESS_TOKEN || ACCESS_TOKEN.indexOf("PASTE_") === 0) {
    console.error("[RankCh] Chua dien ACCESS_TOKEN");
    return;
  }
  if (!CHARACTER_ID || CHARACTER_ID.indexOf("PASTE_") === 0) {
    console.error("[RankCh] Chua dien CHARACTER_ID");
    return;
  }

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

  async function rpc(name, body) {
    const t0 = performance.now();
    const res = await fetch(BASE + "/rest/v1/rpc/" + name, {
      method: "POST",
      headers: {
        apikey: API_KEY,
        authorization: "Bearer " + ACCESS_TOKEN,
        "content-profile": "public",
        "content-type": "application/json",
        "x-client-info": "rankch-console-test/2.0",
      },
      body: JSON.stringify(body),
    });
    const text = await res.text();
    let data = null;
    try {
      data = text ? JSON.parse(text) : null;
    } catch (e) {
      data = { raw: text };
    }
    return {
      httpOk: res.ok,
      status: res.status,
      data: data,
      ms: Math.round(performance.now() - t0),
    };
  }

  console.log("%c[RankCh] 1) leaderboard...", "color:#6db3ff;font-weight:bold");
  const lb = await rpc("rpc_sect_rank_leaderboard", {
    p_character_id: CHARACTER_ID,
    p_min_level: 1,
    p_max_level: 99,
    p_limit: 20,
  });

  if (!lb.httpOk) {
    console.error("[RankCh] leaderboard HTTP FAIL", lb.status, lb.data);
    return;
  }

  const rows = Array.isArray(lb.data)
    ? lb.data
    : Array.isArray(lb.data && lb.data.rows)
      ? lb.data.rows
      : [];

  const realSlotCount = rows.filter(function (r) {
    return r && r.slot != null && Number(r.slot) >= 1;
  }).length;

  console.log(
    "[RankCh] net=" +
      lb.ms +
      "ms · rows=" +
      rows.length +
      " · co field slot that=" +
      realSlotCount +
      "/" +
      rows.length
  );

  console.table(
    rows.slice(0, 15).map(function (r, idx) {
      return {
        api_slot: r.slot != null ? r.slot : "(khong co)",
        fake_index_slot: idx + 1,
        name: r.name,
        atk: r.atk,
        character_id: r.character_id,
      };
    })
  );

  // Chon slot challenge
  let slotsToTry = [];
  if (realSlotCount > 0) {
    // Chi dung slot that, uu tien atk thap / NPC
    const self = String(CHARACTER_ID);
    slotsToTry = rows
      .filter(function (r) {
        return r && r.slot != null && Number(r.slot) >= 1 && String(r.character_id) !== self;
      })
      .map(function (r) {
        return {
          slot: Number(r.slot),
          name: r.name,
          character_id: r.character_id,
          atk: r.atk,
        };
      })
      .sort(function (a, b) {
        const ea = /\[NPC\]|\[Trấn\]|NPC|Lính/i.test(String(a.name || "")) ? 0 : 1;
        const eb = /\[NPC\]|\[Trấn\]|NPC|Lính/i.test(String(b.name || "")) ? 0 : 1;
        if (ea !== eb) return ea - eb;
        return Number(a.atk || 0) - Number(b.atk || 0);
      });
    console.log("[RankCh] Dung SLOT THAT tu API · so luong=" + slotsToTry.length);
  } else {
    // KHONG bi a slot tu index — probe 2..10 (win that: slot 2,3)
    slotsToTry = PROBE_SLOTS.map(function (s) {
      return { slot: s, name: "probe#" + s, character_id: "", atk: null };
    });
    console.warn(
      "[RankCh] API khong tra slot → PROBE slot " +
        PROBE_SLOTS.join(",") +
        " board=" +
        BOARD_CODE +
        " (dung bia 11/14/15)"
    );
  }

  if (!slotsToTry.length) {
    console.warn("[RankCh] Khong co slot de thu");
    return;
  }

  console.log(
    "%c[RankCh] 2) challenge board=" + BOARD_CODE + " · max " + MAX_FIGHTS + " tran",
    "color:#6db3ff;font-weight:bold"
  );

  const hunt = [];
  const results = [];
  let remaining = null;
  let fought = 0;

  for (let i = 0; i < slotsToTry.length && fought < MAX_FIGHTS; i++) {
    const t = slotsToTry[i];
    const slot = Number(t.slot);
    console.log("[RankCh] → slot " + slot + " · " + (t.name || "?") + " · atk " + t.atk);

    const ch = await rpc("rpc_sect_rank_challenge", {
      p_character_id: CHARACTER_ID,
      p_board_code: BOARD_CODE,
      p_target_slot: slot,
    });

    const d = ch.data || {};
    // invalid_slot: thuong HTTP 200 + ok:false
    if (!ch.httpOk || d.ok === false) {
      const reason = d.reason || d.error || d.message || ch.status;
      console.error("[RankCh] FAIL slot " + slot, reason, d);
      results.push({
        n: results.length + 1,
        slot: slot,
        ok: false,
        win: false,
        name: t.name,
        remaining_today: d.remaining_today != null ? d.remaining_today : null,
        err: String(reason).slice(0, 60),
      });
      if (String(reason).indexOf("invalid_slot") >= 0) {
        // thu slot tiep
        await sleep(400);
        continue;
      }
      if (/remaining|limit|daily/i.test(String(reason))) {
        console.warn("[RankCh] Het luot hom nay?");
        break;
      }
      await sleep(DELAY_MS);
      continue;
    }

    fought += 1;
    const win =
      d.win === true ||
      (d.battle && d.battle.winner === "attacker") ||
      (d.battle && d.battle.final && Number(d.battle.final.defender_hp) === 0);
    remaining = d.remaining_today != null ? Number(d.remaining_today) : remaining;
    const defId = (d.defender && d.defender.character_id) || t.character_id || "";
    const defName = (d.defender && d.defender.name) || t.name || "";

    if (d.board_code) {
      console.log("[RankCh] response board_code=" + d.board_code);
    }

    if (win) {
      hunt.push({ id: defId, name: defName, slot: d.target_slot != null ? d.target_slot : slot });
      console.log(
        "%c[RankCh] WIN slot " + slot + " · " + defName + " · remaining_today=" + remaining,
        "color:#3dd68c;font-weight:bold"
      );
    } else {
      console.log(
        "%c[RankCh] LOSE slot " + slot + " · " + defName + " · remaining_today=" + remaining,
        "color:#f5a524"
      );
    }

    results.push({
      n: results.length + 1,
      slot: slot,
      ok: true,
      win: win,
      name: defName,
      defender_id: defId,
      remaining_today: remaining,
      rank_changed: d.rank_changed,
      err: "",
    });

    if (remaining === 0) {
      console.log("[RankCh] remaining_today=0 — het luot");
      break;
    }
    await sleep(DELAY_MS);
  }

  console.log("%c[RankCh] SUMMARY", "color:#6db3ff;font-weight:bold");
  console.table(results);
  console.log("[RankCh] remaining_today =", remaining);
  console.log("[RankCh] hunt (luu id thang):", hunt);
  console.log(`
===== GIAI THICH =====
• invalid_slot = p_target_slot khong hop le tren board nay
• Leaderboard khong co "slot" → KHONG duoc dung hang 11/14/15 trong bang
• Win that cua ban: board=na, target_slot=2 hoac 3, remaining_today giam
• Neu PROBE 2..10 van invalid_slot → sai BOARD_CODE (thu lk/tc/kd/na)
• WIN + remaining_today giam = API danh OK → tool lite se chay duoc
`);
})();

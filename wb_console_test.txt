(async () => {
  // ---------- CONFIG: SUA 3 DONG NAY ----------
  const ACCESS_TOKEN = "PASTE_BEARER_TOKEN_HERE";
  const CHARACTER_ID = "PASTE_CHARACTER_UUID_HERE";
  const TIER = ""; // "" = auto my_tier; hoac "lk" "tc" "kd" "na" "ht" "lh"
  // -------------------------------------------

  const ONLY_IF_OPEN = true; // boss chua ra thi dung sau khi check channels
  const API_KEY = "sb_publishable_vNnNBJooTMczVrWP7qCnhA_479q9nKB";
  const BASE = "https://jeassefmlprfnlszgvbs.supabase.co";
  const MAX_HITS = 20;
  // "game_cd" = cho theo cooldown_sec (giong game, ~3s)
  // "fixed_1500" | "fixed_3000"
  const MODE = "game_cd";

  if (!ACCESS_TOKEN || ACCESS_TOKEN.includes("PASTE_")) {
    console.error("[WB TEST] Chua dien ACCESS_TOKEN");
    return;
  }
  if (!CHARACTER_ID || CHARACTER_ID.includes("PASTE_")) {
    console.error("[WB TEST] Chua dien CHARACTER_ID");
    return;
  }

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

  async function rpc(name, body) {
    const t0 = performance.now();
    const res = await fetch(BASE + "/rest/v1/rpc/" + name, {
      method: "POST",
      headers: {
        apikey: API_KEY,
        authorization: "Bearer " + ACCESS_TOKEN,
        "content-profile": "public",
        "content-type": "application/json",
        "x-client-info": "wb-console-test/1.1",
      },
      body: JSON.stringify(body),
    });
    const text = await res.text();
    let data = null;
    try {
      data = text ? JSON.parse(text) : null;
    } catch (e) {
      data = { raw: text };
    }
    const ms = Math.round(performance.now() - t0);
    const ok = res.ok && !(data && data.ok === false);
    return { ok: ok, status: res.status, data: data, ms: ms };
  }

  console.log("%c[WB TEST] rpc_wb_channels ...", "color:#6db3ff");
  const chRes = await rpc("rpc_wb_channels", { p_character_id: CHARACTER_ID });
  if (!chRes.ok) {
    console.error("[WB TEST] channels FAIL — token het han / sai character?", chRes.status, chRes.data);
    return;
  }

  const myTier = String((chRes.data && chRes.data.my_tier) || "").toLowerCase();
  const channels = Array.isArray(chRes.data && chRes.data.channels) ? chRes.data.channels : [];
  const wantTier = String(TIER || "").toLowerCase();

  let pick = null;
  if (wantTier) {
    for (let i = 0; i < channels.length; i++) {
      if (channels[i].tier === wantTier) {
        pick = channels[i];
        break;
      }
    }
  }
  if (!pick) {
    for (let i = 0; i < channels.length; i++) {
      if (channels[i].available === true) {
        pick = channels[i];
        break;
      }
    }
  }
  if (!pick) {
    for (let i = 0; i < channels.length; i++) {
      if (channels[i].tier === myTier) {
        pick = channels[i];
        break;
      }
    }
  }
  if (!pick && channels.length) pick = channels[0];

  const tier = String((pick && pick.tier) || wantTier || myTier || "lk").toLowerCase();
  const winOpen = pick && pick.window_open === true;
  const hp = pick ? pick.hp_current : null;

  console.table(
    channels.map(function (c) {
      return {
        tier: c.tier,
        available: c.available,
        window_open: c.window_open,
        status: c.status,
        hp_current: c.hp_current,
        hp_max: c.hp_max,
      };
    })
  );

  console.log(
    "[WB TEST] my_tier=" +
      myTier +
      " fight=" +
      tier +
      " window_open=" +
      winOpen +
      " hp=" +
      hp +
      " net=" +
      chRes.ms +
      "ms MODE=" +
      MODE
  );

  // Boss chua ra: van check duoc channels; khong spam attack neu ONLY_IF_OPEN
  if (!winOpen) {
    console.warn(
      "[WB TEST] Boss chua mo cua (window_open=false / status idle). " +
        "Channels VAN chay OK. Khong attack them (ONLY_IF_OPEN=" +
        ONLY_IF_OPEN +
        ")."
    );
    if (ONLY_IF_OPEN) {
      console.log("[WB TEST] STOP — doi boss ra roi chay lai de do gap_ms.");
      return;
    }
    console.warn("[WB TEST] ONLY_IF_OPEN=false — van thu attack (se FAIL nhieu)");
  }

  const rows = [];
  let lastEnd = 0;
  let ok = 0;
  let fail = 0;

  for (let n = 1; n <= MAX_HITS; n++) {
    const tStart = performance.now();
    const gapFromPrev = lastEnd ? Math.round(tStart - lastEnd) : 0;

    const atk = await rpc("rpc_wb_attack", {
      p_character_id: CHARACTER_ID,
      p_tier: tier,
    });
    const tEnd = performance.now();
    lastEnd = tEnd;

    const d = atk.data || {};
    const success = atk.ok === true;
    if (success) ok++;
    else fail++;

    const cdSec = Number(d.cooldown_sec != null ? d.cooldown_sec : d.atk_speed_sec != null ? d.atk_speed_sec : 3) || 3;
    const serverCdMs = Math.round(cdSec * 1000);
    let waitMs = serverCdMs;
    if (MODE === "fixed_1500") waitMs = 1500;
    else if (MODE === "fixed_3000") waitMs = 3000;
    else {
      const elapsed = tEnd - tStart;
      waitMs = Math.max(0, serverCdMs - elapsed);
    }

    let errMsg = "";
    if (!success) {
      errMsg = String(d.error || d.reason || d.message || atk.status || "fail").slice(0, 80);
    }

    const row = {
      n: n,
      ok: success,
      net_ms: atk.ms,
      gap_ms: gapFromPrev,
      dmg: d.damage,
      hp_after: d.hp_after,
      cd_sec: d.cooldown_sec,
      killed: d.killed,
      err: errMsg,
      next_wait_ms: Math.round(waitMs),
    };
    rows.push(row);

    console.log(
      (success ? "[OK] " : "[FAIL] ") +
        "#" +
        n +
        " net=" +
        atk.ms +
        "ms gap=" +
        gapFromPrev +
        "ms dmg=" +
        (d.damage != null ? d.damage : "-") +
        " hp=" +
        (d.hp_after != null ? d.hp_after : "-") +
        " wait=" +
        Math.round(waitMs) +
        "ms " +
        errMsg
    );

    if (d.killed === true || (Number(d.hp_after) === 0 && d.hp_after != null)) {
      console.log("[WB TEST] BOSS DEAD — stop");
      break;
    }

    if (n < MAX_HITS) await sleep(waitMs);
  }

  console.table(rows);

  const okRows = rows.filter(function (r) {
    return r.ok;
  });
  const gaps = okRows
    .map(function (r) {
      return r.gap_ms;
    })
    .filter(function (g) {
      return g > 0;
    });
  let avgGap = 0;
  if (gaps.length) {
    let s = 0;
    for (let i = 0; i < gaps.length; i++) s += gaps[i];
    avgGap = Math.round(s / gaps.length);
  }
  let avgNet = 0;
  if (okRows.length) {
    let s = 0;
    for (let i = 0; i < okRows.length; i++) s += okRows[i].net_ms;
    avgNet = Math.round(s / okRows.length);
  }

  console.log(
    "[WB TEST] SUMMARY total=" +
      rows.length +
      " ok=" +
      ok +
      " fail=" +
      fail +
      " avg_gap_ok=" +
      avgGap +
      "ms avg_net=" +
      avgNet +
      "ms"
  );
  console.log(
    "[WB TEST] Doc: game_cd → gap ~3000ms it FAIL. fixed_1500 → nhieu FAIL cooldown."
  );
})();
